# MUST be UTF-8. The version this replaced was Windows-1255 (cp1255), which
# silently voided the Hebrew rule below — robots.txt is defined as UTF-8, so
# non-UTF-8 bytes are undefined and ignored. Upload in FileZilla's BINARY mode,
# never ASCII, or the Hebrew will be mangled again just as silently.

User-agent: *
# Entry point and action URLs. The previous rule was "/index.php?", which only
# matched URLs that carried a query string — so bare /index.php was allowed, and
# it became the single most-crawled path on the site. Without the "?" this
# prefix covers the bare path and every query form.
Disallow: /index.php
Disallow: /api.php
# Special-page namespace, both spellings. Hebrew is canonical here and is what
# crawlers actually request: מיוחד:Drilldown/... generates faceted filter
# combinations, i.e. an effectively infinite URL space. The percent-encoded form
# is belt-and-braces for crawlers that do not normalise UTF-8 paths themselves.
Disallow: /Special:
Disallow: /מיוחד:
Disallow: /%D7%9E%D7%99%D7%95%D7%97%D7%93:
Crawl-delay: 10
